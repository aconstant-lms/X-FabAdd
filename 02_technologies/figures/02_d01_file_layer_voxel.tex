% 02_d01  from shape to layer to path to voxel (C-shaped part, oblique view)
\documentclass[tikz,border=8pt]{standalone}
\usepackage{xfab-tikz}
\begin{document}
% oblique projection of the xy-plane: x -> (1,0), y -> (0.45,0.32)
\tikzset{plane/.style={cm={1,0,0.45,0.32,(0,#1)}}}
\def\Ro{1.25}\def\Ri{0.6}
% C-shape outline in its plane (opening towards +y): outer arc 180..360, inner arc back
\def\Cpath{(-\Ro,0) arc[start angle=180,end angle=360,radius=\Ro] -- (\Ri,0) arc[start angle=360,end angle=180,radius=\Ri] -- cycle}
% projected point helper: (x,y,z) -> (x+0.45y, 0.32y+z)
\def\P(#1,#2,#3){({#1+0.45*(#2)},{0.32*(#2)+#3})}
\begin{tikzpicture}[x=1cm,y=1cm]

% ---------- 1 shape: CAD file (height H = 1.3) with STL facets and vertices
\begin{scope}[shift={(1.4,0.4)}]
  \def\H{1.3}
  % outer wall, front-facing (angles 180..360), as one band
  \fill[fill=cPart!35, draw=cPartD, line width=0.7pt]
    plot[domain=180:360, samples=40] ({\Ro*cos(\x)+0.45*\Ro*sin(\x)},{0.32*\Ro*sin(\x)+0})
    -- plot[domain=360:180, samples=40] ({\Ro*cos(\x)+0.45*\Ro*sin(\x)},{0.32*\Ro*sin(\x)+\H}) -- cycle;
  % end faces (visible, in the plane y = 0)
  \fill[fill=cPart!55, draw=cPartD, line width=0.7pt] \P(\Ri,0,0) -- \P(\Ro,0,0) -- \P(\Ro,0,\H) -- \P(\Ri,0,\H) -- cycle;
  % top face
  \begin{scope}[plane=\H] \fill[fill=cPart!20, draw=cPartD, line width=0.7pt] \Cpath; \end{scope}
  % triangle facets on the outer wall and the top face
  \foreach \a in {180,202.5,...,337.5}{
    \pgfmathsetmacro{\b}{\a+22.5}
    \draw[cPartD, line width=0.3pt] \P({\Ro*cos(\a)},{\Ro*sin(\a)},0) -- \P({\Ro*cos(\a)},{\Ro*sin(\a)},\H)
                                   -- \P({\Ro*cos(\b)},{\Ro*sin(\b)},0);
    \draw[cPartD, line width=0.3pt] \P({\Ro*cos(\a)},{\Ro*sin(\a)},\H) -- \P({\Ri*cos(\a)},{\Ri*sin(\a)},\H)
                                   -- \P({\Ro*cos(\b)},{\Ro*sin(\b)},\H);}
  \foreach \a in {180,202.5,...,360}{
    \fill[cPartD] \P({\Ro*cos(\a)},{\Ro*sin(\a)},0) circle (0.035);
    \fill[cPartD] \P({\Ro*cos(\a)},{\Ro*sin(\a)},\H) circle (0.035);
    \fill[cPartD] \P({\Ri*cos(\a)},{\Ri*sin(\a)},\H) circle (0.035);}
  \node[labc, anchor=north] at (0.25,-1.0) {1\enspace shape\\CAD file (STL facets)};
\end{scope}
\draw[stepto] (3.6,0.9) -- (4.3,0.9);

% ---------- 2 layer, thickness dz
\begin{scope}[shift={(6.0,0.6)}]
  \def\H{0.22}
  \fill[fill=cPart!35, draw=cPartD, line width=0.7pt]
    plot[domain=180:360, samples=40] ({\Ro*cos(\x)+0.45*\Ro*sin(\x)},{0.32*\Ro*sin(\x)+0})
    -- plot[domain=360:180, samples=40] ({\Ro*cos(\x)+0.45*\Ro*sin(\x)},{0.32*\Ro*sin(\x)+\H}) -- cycle;
  \fill[fill=cPart!55, draw=cPartD, line width=0.7pt] \P(\Ri,0,0) -- \P(\Ro,0,0) -- \P(\Ro,0,\H) -- \P(\Ri,0,\H) -- cycle;
  \begin{scope}[plane=\H] \fill[fill=cPart!20, draw=cPartD, line width=0.7pt] \Cpath; \end{scope}
  \draw[dim] (-1.55,0) -- (-1.55,\H); \draw[cPartD, line width=0.3pt] (-1.6,0) -- (-1.3,0) (-1.6,\H) -- (-1.3,\H);
  \node[lab, anchor=east] at (-1.6,0.11) {$\Delta z$};
  \node[labc, anchor=north] at (0.25,-1.2) {2\enspace layer\\slice of thickness $\Delta z$};
\end{scope}
\draw[stepto] (8.2,0.9) -- (8.9,0.9);

% ---------- 3 path: contour + hatching of the layer
\begin{scope}[shift={(10.6,0.6)}]
  \def\H{0.22}
  \fill[fill=cPart!35, draw=cPartD, line width=0.7pt]
    plot[domain=180:360, samples=40] ({\Ro*cos(\x)+0.45*\Ro*sin(\x)},{0.32*\Ro*sin(\x)+0})
    -- plot[domain=360:180, samples=40] ({\Ro*cos(\x)+0.45*\Ro*sin(\x)},{0.32*\Ro*sin(\x)+\H}) -- cycle;
  \fill[fill=cPart!55, draw=cPartD, line width=0.7pt] \P(\Ri,0,0) -- \P(\Ro,0,0) -- \P(\Ro,0,\H) -- \P(\Ri,0,\H) -- cycle;
  \begin{scope}[plane=\H]
    \fill[fill=white, draw=cPartD, line width=0.7pt] \Cpath;
    % contour (offset inwards)
    \draw[cLight, line width=0.9pt] (-\Ro+0.1,0) arc[start angle=180,end angle=360,radius=\Ro-0.1] -- (\Ri+0.1,0)
          arc[start angle=360,end angle=180,radius=\Ri+0.1] -- cycle;
    % hatching, clipped to the inner region
    \begin{scope}
      \clip (-\Ro+0.2,0) arc[start angle=180,end angle=360,radius=\Ro-0.2] -- (\Ri+0.2,0)
            arc[start angle=360,end angle=180,radius=\Ri+0.2] -- cycle;
      \foreach \xx in {-1.3,-1.2,...,1.3} \draw[cLight, line width=0.5pt] (\xx,-1.4) -- (\xx,0.1);
    \end{scope}
  \end{scope}
  \node[labc, anchor=north] at (0.25,-1.2) {3\enspace path\\contour + layer hatching};
\end{scope}
\draw[stepto] (12.8,0.9) -- (13.5,0.9);

% ---------- 4 voxel: the spot
\begin{scope}[shift={(14.6,0.75)}]
  \def\s{0.5}
  \fill[fill=cPart, draw=cPartD, line width=0.7pt] (0,0) rectangle (\s,\s);
  \fill[fill=cPart!60, draw=cPartD, line width=0.7pt] (0,\s) -- ++(0.45*\s,0.32*\s) -- ++(\s,0) -- (\s,\s) -- cycle;
  \fill[fill=cPartD!80, draw=cPartD, line width=0.7pt] (\s,0) -- ++(0.45*\s,0.32*\s) -- ++(0,\s) -- (\s,\s) -- cycle;
  \draw[lighthalo] (0.36,1.55) -- (0.36,0.75); \draw[light] (0.36,1.55) -- (0.36,0.75);
  \node[labc, anchor=north] at (0.35,-1.05) {4\enspace voxel\\spot $\times$ track $\times$ $\Delta z$};
\end{scope}
\xfabref{0}{-2.1}{The part is no longer stored in a mould but in a file: shape, then layers, then paths, then voxels.}
\end{tikzpicture}
\end{document}
